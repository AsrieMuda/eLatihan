2. Kod Backend Google Apps Script (Code.gs)

const SPREADSHEET_ID = '1uByxXITmcC2AN-nycXPpmVup8jbdbcRnuYwAo-SPk2A';
const FOLDER_NAME = 'ARKIB_SIJIL_LATIHAN_ANGGOTA';

function doGet(e) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheetStaff = ss.getSheetByName('HARI BERKURSUS') || ss.getSheets()[0];
  const staffRows = sheetStaff.getDataRange().getValues();
  
  const staff = [];
  for (let i = 3; i < staffRows.length; i++) {
    if (staffRows[i][1]) {
      staff.push({
        id: i + 1,
        bil: staffRows[i][0] || (i - 2),
        nama: String(staffRows[i][1]).trim(),
        jawatan: String(staffRows[i][2] || '').trim(),
        skim: String(staffRows[i][3] || 'SOK 1').trim(),
        hari: parseFloat(staffRows[i][4]) || 0,
        email: String(staffRows[i][5] || '').trim(),
        role: String(staffRows[i][6] || 'USER').trim()
      });
    }
  }

  let sheetLogs = ss.getSheetByName('REKOD_KURSUS');
  const logs = [];
  if (sheetLogs) {
    const logRows = sheetLogs.getDataRange().getValues();
    for (let j = 1; j < logRows.length; j++) {
      if (logRows[j][1]) {
        logs.push({
          id: logRows[j][0],
          nama: logRows[j][1],
          tajuk: logRows[j][2],
          penganjur: logRows[j][3],
          kategori: logRows[j][4],
          tarikh: logRows[j][5],
          hari: parseFloat(logRows[j][6]) || 0,
          sijilUrl: logRows[j][7] || ''
        });
      }
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ status: 'success', staff, logs }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

  if (data.action === 'sendOTP') {
    const otp = Math.floor(100000 + Math.random() * 900000).toString();
    CacheService.getScriptCache().put('OTP_' + data.email, otp, 300);
    MailApp.sendEmail({
      to: data.email,
      subject: 'Kod Pengesahan 2FA (OTP) - e-Latihan BKKM',
      body: 'Salam Sejahtera ' + data.nama + ',\n\nKod OTP keselamatan anda ialah: ' + otp + '\nKod ini sah selama 5 minit.'
    });
    return ContentService.createTextOutput(JSON.stringify({ status: 'otp_sent' })).setMimeType(ContentService.MimeType.JSON);
  }

  if (data.action === 'verifyOTP') {
    const cached = CacheService.getScriptCache().get('OTP_' + data.email);
    const valid = (cached && cached === data.otp);
    return ContentService.createTextOutput(JSON.stringify({ status: valid ? 'verified' : 'invalid' })).setMimeType(ContentService.MimeType.JSON);
  }
  
  if (data.action === 'addCourse' || data.action === 'updateCourse') {
    let sijilUrl = data.existingSijilUrl || '';
    if (data.fileBase64 && data.fileName) {
      const folders = DriveApp.getFoldersByName(FOLDER_NAME);
      const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(FOLDER_NAME);
      const blob = Utilities.newBlob(Utilities.base64Decode(data.fileBase64), data.mimeType, data.nama + '_' + data.fileName);
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      sijilUrl = file.getUrl();
    }

    let sheetLogs = ss.getSheetByName('REKOD_KURSUS');
    if (!sheetLogs) {
      sheetLogs = ss.insertSheet('REKOD_KURSUS');
      sheetLogs.appendRow(['ID', 'NAMA', 'TAJUK KURSUS', 'PENGANJUR', 'KATEGORI', 'TARIKH', 'HARI', 'PAUTAN SIJIL DRIVE']);
    }

    if (data.action === 'updateCourse' && data.courseId) {
      const logRows = sheetLogs.getDataRange().getValues();
      for (let r = 1; r < logRows.length; r++) {
        if (String(logRows[r][0]) === String(data.courseId)) {
          sheetLogs.getRange(r + 1, 3, 1, 6).setValues([[data.tajuk, data.penganjur, data.kategori, data.tarikh, data.hari, sijilUrl]]);
          break;
        }
      }
    } else {
      sheetLogs.appendRow([new Date().getTime(), data.nama, data.tajuk, data.penganjur, data.kategori, data.tarikh, data.hari, sijilUrl]);
    }

    const sheetStaff = ss.getSheetByName('HARI BERKURSUS') || ss.getSheets()[0];
    const rows = sheetStaff.getDataRange().getValues();
    const deltaDays = parseFloat(data.deltaDays !== undefined ? data.deltaDays : data.hari) || 0;
    for (let i = 3; i < rows.length; i++) {
      if (String(rows[i][1]).trim().toUpperCase() === String(data.nama).trim().toUpperCase()) {
        const currentDays = parseFloat(rows[i][4]) || 0;
        sheetStaff.getRange(i + 1, 5).setValue(Math.max(0, currentDays + deltaDays));
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'success', sijilUrl })).setMimeType(ContentService.MimeType.JSON);
  }
}

3. Langkah Pemasangan Selepas Ekstrak ZIP

  1. Backend: Buka Google Sheet → Extensions → Apps Script → tampal backend/Code.gs → Deploy → Web App (Anyone).
  2. Frontend: Muat naik isi folder frontend/ ke repositori GitHub dan aktifkan Settings → Pages.
  3. Integrasi: Tampal URL Web App pada Langkah 1 di atas dan klik Simpan & Sambung.
